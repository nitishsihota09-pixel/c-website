#include <stdio.h>
#include <conio.h>

void change(int *num3, int *num4)
{
    int value;
    value = *num3;
    *num3 = *num4;
    *num4 = value;
}

void main()
{
    int num1, num2;
    clrscr();

    printf("enter num1 :- ");
    scanf("%d", &num1);

    printf("enter num2 :- ");
    scanf("%d", &num2);

    change(&num1, &num2);

    printf("change1 :- %d, change2 :- %d", num1, num2);
    
}
Output : 
enter num1 :- 10
enter num2 :- 20

change1 :- 20, change2 :- 10
