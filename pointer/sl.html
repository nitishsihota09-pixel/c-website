#include <stdio.h>
#include <conio.h>

void swap(int *num1, int *num2)
{
    int num;

    num = *num1;
    *num1= *num2;
    *num2= num;

}
void main(){
    int num1,num2;
    printf("Enter first numbers: ");
    scanf("%d",&num1);

    printf("Enter Second numbers: ");
    scanf("%d",&num2);

    swap(&num1, &num2);

    printf("Now first number is :- %d", num1);
    printf("Now Second number is :- %d", num2);


}
Output : 
        
Enter first numbers: 10
Enter Second numbers: 20

Now first number is :- 20
Now Second number is :- 10
