#include <stdio.h>
#include <conio.h>

void main()
{
    int array[5],i,j,value;
    clrscr();

    printf("enter five values :-");

    for(i=0;i<=4;i++)
    {
        scanf("%d",&array[i]);
    }

    for(i=0;i<=4;i++)
    {
        for(j=i;j<=4;j++)
        {
            if(array[i]>array[j])
            {
                value=array[i];
                array[i]=array[j];
                array[j]=value;
            }
        }
    }

    for(i=0;i<=4;i++)
    {
        printf("%d",array[i]);
    }
 
}
Output : 
enter five values :- 8 3 10 2 5

2 3 5 8 10
