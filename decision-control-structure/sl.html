#include <stdio.h>
#include <conio.h>
void main()
{
    
    int bdate,bmonth,byear,cdate,cmonth,cyear;
    int date,month,year;

    clrscr();

    printf("Enter birthday date: ");
    scanf("%d %d %d",&bdate,&bmonth,&byear);

    printf("Enter current date: ");
    scanf("%d %d %d",&cdate,&cmonth,&cyear);

    year=cyear-byear;
    month=cmonth-bmonth;
    date=cdate-bdate;

    printf("Difference %d day %d month %d year",date,month,year);
}

Enter birthday date: 10 5 2006
Enter current date: 10 5 2026
