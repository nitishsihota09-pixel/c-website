#include <stdio.h>
#include <conio.h>
void main()
{
   int matrix1[2][2];
   int matrix2[2][2];
   int matrix3[2][2];
   int i,j;
  clrscr();

  printf("Enter Value of First Matrix :-\n")
  for(i=0;i<=1;i++){
    for(j=0;j<=1;j++){
      printf("Enter no. :- ");
      scanf("%d",&matrix1[i][j]);
    }
  }

  printf("Enter Value of Second Matrix :-\n")
  for(i=0;i<=1;i++){
    for(j=0;j<=1;j++){
      printf("Enter no. :- ");
      scanf("%d",&matrix2[i][j]);
    }
  }

   printf("Addition of 2 Matrix :-");
   for(i=0;i<2;i++){
    for(j=0;j<2;j++){
       matrix3[i][j]=matrix1[i][j]+matrix2[i][j];
      printf("%d",matrix3[j][i]);
    }
    printf("\n");
}
}
Output : 
Enter Value of First Matrix :-
Enter no. :- 1
Enter no. :- 2
Enter no. :- 3
Enter no. :- 4

Enter Value of Second Matrix :-
Enter no. :- 5
Enter no. :- 6
Enter no. :- 7
Enter no. :- 8

Addition of 2 Matrix :- 6 10
                        8 12
